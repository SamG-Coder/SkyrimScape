cmake_minimum_required(VERSION 3.24)
project(SkyrimScape VERSION 0.2.6 LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(SKYRIMSCAPE_BUILD_PLUGIN "Build the Windows SKSE plugin (requires CommonLibSSE-NG)" ON)
if(SKYRIMSCAPE_BUILD_PLUGIN)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
add_subdirectory(external/CommonLibSSE-NG)
include(external/CommonLibSSE-NG/cmake/CommonLibSSE.cmake)
add_commonlibsse_plugin(SkyrimScape NAME SkyrimScape AUTHOR SamGCoder VERSION 0.2.6 USE_ADDRESS_LIBRARY SOURCES src/plugin.cpp)
target_compile_features(SkyrimScape PRIVATE cxx_std_23)
target_precompile_headers(SkyrimScape PRIVATE src/pch.hpp)
endif()
enable_testing()
add_executable(SkyrimScapeControlTests tests/control_tests.cpp)
target_compile_features(SkyrimScapeControlTests PRIVATE cxx_std_20)
target_include_directories(SkyrimScapeControlTests PRIVATE src)
add_test(NAME control_math COMMAND SkyrimScapeControlTests)
add_executable(SkyrimScapeNavigationTests tests/navigation_tests.cpp)
target_compile_features(SkyrimScapeNavigationTests PRIVATE cxx_std_20)
target_include_directories(SkyrimScapeNavigationTests PRIVATE src)
add_test(NAME navigation COMMAND SkyrimScapeNavigationTests)
add_executable(SkyrimScapeRouteReplay scripts/replay_route.cpp)
target_compile_features(SkyrimScapeRouteReplay PRIVATE cxx_std_20)
target_include_directories(SkyrimScapeRouteReplay PRIVATE src)
