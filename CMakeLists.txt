cmake_minimum_required(VERSION 3.24)
project(SkyrimScape VERSION 0.3.33 LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(SKYRIMSCAPE_BUILD_PLUGIN "Build the Windows SKSE plugin (requires CommonLibSSE-NG)" ON)
if(SKYRIMSCAPE_BUILD_PLUGIN)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
add_subdirectory(external/CommonLibSSE-NG)
include(external/CommonLibSSE-NG/cmake/CommonLibSSE.cmake)
set(MESHOPT_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(external/meshoptimizer)
add_commonlibsse_plugin(SkyrimScape NAME SkyrimScape AUTHOR SamGCoder VERSION 0.3.33 USE_ADDRESS_LIBRARY SOURCES src/plugin.cpp src/runtime_style.cpp src/anime_render.cpp)
target_compile_features(SkyrimScape PRIVATE cxx_std_23)
target_link_libraries(SkyrimScape PRIVATE gdi32 d3d11 d3dcompiler meshoptimizer)
target_precompile_headers(SkyrimScape PRIVATE src/pch.hpp)
endif()
enable_testing()
add_executable(SkyrimScapeRoofTests tests/roof_tests.cpp)
target_compile_features(SkyrimScapeRoofTests PRIVATE cxx_std_20)
target_include_directories(SkyrimScapeRoofTests PRIVATE src)
add_test(NAME roof_policy COMMAND SkyrimScapeRoofTests)
add_executable(SkyrimScapeControlTests tests/control_tests.cpp)
target_compile_features(SkyrimScapeControlTests PRIVATE cxx_std_20)
target_include_directories(SkyrimScapeControlTests PRIVATE src)
add_test(NAME control_math COMMAND SkyrimScapeControlTests)
add_executable(SkyrimScapeNavigationTests tests/navigation_tests.cpp)
target_compile_features(SkyrimScapeNavigationTests PRIVATE cxx_std_20)
target_include_directories(SkyrimScapeNavigationTests PRIVATE src)
add_test(NAME navigation COMMAND SkyrimScapeNavigationTests)
add_executable(SkyrimScapeRouteReplay scripts/replay_route.cpp)
target_compile_features(SkyrimScapeRouteReplay PRIVATE cxx_std_20)
target_include_directories(SkyrimScapeRouteReplay PRIVATE src)
add_executable(SkyrimScapeGridTests tests/grid_tests.cpp)
target_compile_features(SkyrimScapeGridTests PRIVATE cxx_std_20)
target_include_directories(SkyrimScapeGridTests PRIVATE src)
add_test(NAME grid COMMAND SkyrimScapeGridTests)
add_executable(SkyrimScapeCombatTests tests/combat_tests.cpp)
target_compile_features(SkyrimScapeCombatTests PRIVATE cxx_std_20)
target_include_directories(SkyrimScapeCombatTests PRIVATE src)
add_test(NAME combat COMMAND SkyrimScapeCombatTests)
add_executable(SkyrimScapeContextMenuTests tests/context_menu_tests.cpp)
target_compile_features(SkyrimScapeContextMenuTests PRIVATE cxx_std_20)
target_include_directories(SkyrimScapeContextMenuTests PRIVATE src)
add_test(NAME context_menu COMMAND SkyrimScapeContextMenuTests)
if(WIN32)
add_executable(SkyrimScapeSystemFontTests tests/system_font_tests.cpp)
target_compile_features(SkyrimScapeSystemFontTests PRIVATE cxx_std_20)
target_include_directories(SkyrimScapeSystemFontTests PRIVATE src)
target_link_libraries(SkyrimScapeSystemFontTests PRIVATE gdi32)
add_test(NAME system_font COMMAND SkyrimScapeSystemFontTests)
if(SKYRIMSCAPE_BUILD_PLUGIN)
add_executable(SkyrimScapeAnimeTests tests/anime_tests.cpp)
target_compile_features(SkyrimScapeAnimeTests PRIVATE cxx_std_20)
target_include_directories(SkyrimScapeAnimeTests PRIVATE src)
target_link_libraries(SkyrimScapeAnimeTests PRIVATE d3d11 d3dcompiler)
add_test(NAME anime_shader COMMAND SkyrimScapeAnimeTests)
add_executable(SkyrimScapeStyleTests tests/style_tests.cpp)
target_compile_features(SkyrimScapeStyleTests PRIVATE cxx_std_20)
target_include_directories(SkyrimScapeStyleTests PRIVATE src)
target_link_libraries(SkyrimScapeStyleTests PRIVATE d3d11 meshoptimizer)
add_test(NAME runtime_style COMMAND SkyrimScapeStyleTests)
endif()
endif()
